  updateMarkerPosition(lat: number, lng: number) {
    const newPosition = new google.maps.LatLng(lat, lng);
    const currentPos = this.marker.getPosition();
    console.log('marker position ', currentPos.lat());

    const service = new google.maps.DistanceMatrixService();
    service.getDistanceMatrix(
      {
        origins: [newPosition],
        destinations: [this.endPoint],
        travelMode: google.maps.TravelMode.DRIVING,
      },
      (response: any, status: any) => {
        if (status === google.maps.DistanceMatrixStatus.OK) {
          const results = response.rows[0].elements;
          // console.log('Distance: ', results[0].distance.text);
          // console.log('Duration: ', results[0].duration.text);
        } else {
          console.error('Error:', status);
        }
      }
    );

    const steps = 100; // Number of steps for smooth transition
    const stepDuration = 1000 / steps; // Duration per step

    let currentStep = 0;

    const latStep = (lat - currentPos.lat()) / steps;
    const lngStep = (lng - currentPos.lng()) / steps;

    const animateMarker = () => {
      currentStep++;
      const newLat = currentPos.lat() + latStep * currentStep;
      const newLng = currentPos.lng() + lngStep * currentStep;

      this.marker.setPosition({ lat: newLat, lng: newLng });

      if (currentStep < steps) {
        requestAnimationFrame(animateMarker);
      } else {
        this.marker.setAnimation(null);
      }
    };

    animateMarker();
    this.checkPositionOnRoute({ lat, lng });
    this.map.setCenter(newPosition);
  }

  reachedOrigin = false;
  checkPositionOnRoute(position: { lat: number; lng: number }) {
    const locationIconPath =
      'M12 2C8.13 2 5 5.13 5 9c0 5.25 7 13 7 13s7-7.75 7-13c0-3.87-3.13-7-7-7zm0 9.5c-1.38 0-2.5-1.12-2.5-2.5S10.62 6.5 12 6.5 14.5 7.62 14.5 9 13.38 11.5 12 11.5z';
    let onRoute = false;
    const deviationThreshold = 10; // Adjust the threshold in meters

    console.log('routePath list is : ', this.routePath);

    let distance: any;
    let deviation = false;

    const origin = { lat: this.pickuplt, lng: this.pickuplng };

    // when the position is reached to the origin
    if (position.lat === origin.lat && position.lng === origin.lng) {
      this.reachedOrigin = true;
    }

    if (this.reachedOrigin) {
      for (let i = 0; i < this.routePath.length - 1; i++) {
        const segmentStart = this.routePath[i];
        const segmentEnd = this.routePath[i + 1];
        distance = this.distanceToSegment(position, segmentStart, segmentEnd);

        if (distance < deviationThreshold) {
          onRoute = true;
          this.updatePolylineColor(this.routePath[i]);
          break;
        }
      }
      if (!onRoute) {
        deviation = true;
      }
    } else {
      for (let i = 0; i < this.routePath.length - 1; i++) {
        const segmentStart = this.routePath[i];
        const segmentEnd = this.routePath[i + 1];
        distance = this.distanceToSegment(position, segmentStart, segmentEnd);

        if (distance < deviationThreshold) {
          onRoute = true;
          break;
        }
      }
    }

    if (onRoute) {

      // Set the red color for the previous marker to be a standard marker
      new google.maps.Marker({
        position: position,
        map: this.map,
        title: 'Previous Position',
        icon: {
          path: locationIconPath, // Use a location shape
          fillColor: 'green', // Change fillColor to green for previous positions
          fillOpacity: 1,
          scale: 2,
          strokeColor: 'white',
          strokeWeight: 2,
        },
      });
      this.updatePolylineColor(position);


      // Marker is on the route, set to green
      this.marker.setIcon({
        path: this.marker.getIcon().path,
        fillColor: 'green',
        fillOpacity: 1,
        scale: 10,
        strokeColor: 'white',
        strokeWeight: 2,
      });
    } 
    else {

      // Set the red color for the previous marker to be a standard marker
      new google.maps.Marker({
        position: position,
        map: this.map,
        title: 'Previous Position',
        icon: {
          path: locationIconPath, // Use a location shape
          fillColor: 'red', // Change fillColor to red for previous positions
          fillOpacity: 1,
          scale: 2,
          strokeColor: 'white',
          strokeWeight: 2,
        },
      });
      // Marker is off the route, set to red
      this.marker.setIcon({
        path: this.marker.getIcon().path,
        fillColor: 'red',
        fillOpacity: 1,
        scale: 10,
        strokeColor: 'white',
        strokeWeight: 2,
      });

      if (deviation) {
        alert('Marker is out of the deviation!');
      }
    }
  }

  updatePolylineColor(position: any) {
    console.log('reached on the route');
    const traveledPath = [];
    const untraveledPath = [];

    let hasReached = false;
    for (const point of this.routePath) {
      if (!hasReached && this.haversineDistance(position, point) < 1) {
        hasReached = true;
      }

      if (hasReached) {
        untraveledPath.push(point);
      } else {
        traveledPath.push(point);
      }
    }

    const traveledPolyline = new google.maps.Polyline({
      path: traveledPath,
      strokeColor: '#00FF00',
      strokeOpacity: 1.0,
      strokeWeight: 3,
    });

    const untraveledPolyline = new google.maps.Polyline({
      path: untraveledPath,
      strokeColor: '#0000FF',
      strokeOpacity: 1.0,
      strokeWeight: 3,
    });

    traveledPolyline.setMap(this.map);
    untraveledPolyline.setMap(this.map);
  }

  distanceToSegment(
    position: { lat: number; lng: number },
    segmentStart: { lat: number; lng: number },
    segmentEnd: { lat: number; lng: number }
  ): number {
    const lat1 = segmentStart.lat;
    const lon1 = segmentStart.lng;
    const lat2 = segmentEnd.lat;
    const lon2 = segmentEnd.lng;
    const lat3 = position.lat;
    const lon3 = position.lng;

    const A = lat3 - lat1;
    const B = lon3 - lon1;
    const C = lat2 - lat1;
    const D = lon2 - lon1;

    const dot = A * C + B * D;
    const len_sq = C * C + D * D;
    const param = len_sq !== 0 ? dot / len_sq : -1;

    let xx, yy;

    if (param < 0) {
      xx = lat1;
      yy = lon1;
    } else if (param > 1) {
      xx = lat2;
      yy = lon2;
    } else {
      xx = lat1 + param * C;
      yy = lon1 + param * D;
    }

    const dx = lat3 - xx;
    const dy = lon3 - yy;

    return Math.sqrt(dx * dx + dy * dy) * 100000; // Convert to meters
  }

  haversineDistance(
    pos1: { lat: number; lng: number },
    pos2: { lat: number; lng: number }
  ): number {
    const R = 6371; // Radius of the Earth in km
    const dLat = ((pos2.lat - pos1.lat) * Math.PI) / 180;
    const dLng = ((pos2.lng - pos1.lng) * Math.PI) / 180;
    const a =
      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
      Math.cos((pos1.lat * Math.PI) / 180) *
        Math.cos((pos2.lat * Math.PI) / 180) *
        Math.sin(dLng / 2) *
        Math.sin(dLng / 2);
    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
    return R * c * 1000; // Distance in meters
  }

            this.Longitude = resp.coords.longitude;

            console.log('lat long', this.Latitude, this.Longitude);

            this.trackService
              .updatecords(this.id, this.Latitude, this.Longitude)
              .subscribe((data) => {
                console.log('data response', data);
              });
            this.trackService
              .updatecords1(this.id, this.Latitude, this.Longitude)
              .subscribe((data) => {
                console.log('data response1', data);

              .subscribe((data: any) => {

  takePicture = async (card: number) => {
    const image = await Camera.getPhoto({
      quality: 90,
      allowEditing: false,
      resultType: CameraResultType.Uri,
      source: CameraSource.Camera,
      saveToGallery: true,
    });

    this.imageSource[card] = this.domSanitizer.bypassSecurityTrustUrl(
      image.webPath ? image.webPath : ''
    );

    // Convert the captured image to a Blob
    const response = await fetch(image.webPath!);
    const blob = await response.blob();
    this.file = new File([blob], `image-${card}.jpg`, { type: 'image/jpeg' });
  };
