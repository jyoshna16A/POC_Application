  updateMarkerPosition(lat: number, lng: number) {
    const newPosition = new google.maps.LatLng(lat, lng);
    const currentPos = this.marker.getPosition();
    const steps = 100; // Number of steps for smooth transition
    const stepDuration = 1000 / steps; // Duration per step

    let currentStep = 0;

    const latStep = (lat - currentPos.lat()) / steps;
    const lngStep = (lng - currentPos.lng()) / steps;

    const animateMarker = () => {
      currentStep++;
      const lat = currentPos.lat() + latStep * currentStep;
      const lng = currentPos.lng() + lngStep * currentStep;

      this.marker.setPosition({ lat, lng });
      // this.marker.setAnimation(google.maps.Animation.BOUNCE);

      if (currentStep < steps) {
        // requestAnimationFrame(animateMarker);
        setTimeout(animateMarker, stepDuration);
      } else {
        this.marker.setAnimation(null);
      }
    };

    animateMarker();
    this.checkPositionOnRoute({ lat, lng });
    this.map.setCenter(newPosition);
  }
  
  checkPositionOnRoute(position: { lat: number, lng: number }) {
    const locationIconPath =
      'M12 2C8.13 2 5 5.13 5 9c0 5.25 7 13 7 13s7-7.75 7-13c0-3.87-3.13-7-7-7zm0 9.5c-1.38 0-2.5-1.12-2.5-2.5S10.62 6.5 12 6.5 14.5 7.62 14.5 9 13.38 11.5 12 11.5z';
    let onRoute = false;
    const deviationThreshold = 10; // Adjust the threshold in meters

    console.log("routePath list is : ",this.routePath);

    let distance : any;
    let reachedOrigin = false;

    // Check if the marker's position is close to any point on the route
    const origin = this.routePath[0];
    if (this.haversineDistance(position, origin) < deviationThreshold) {
        let reachedOrigin = true;
        for (let i = 0; i < this.routePath.length; i++) {
            distance = this.distanceToSegment(position, this.routePath[i], this.routePath[i + 1] || this.routePath[i]);
            if (this.haversineDistance(position, this.routePath[i]) < deviationThreshold || distance < deviationThreshold) {
                this.updatePolylineColor(position);
                onRoute = true;
                break;
            }
        }
        if (!onRoute) {
            reachedOrigin = false;
        }
    } else {
        // Check the route for the position
        for (let i = 0; i < this.routePath.length - 1; i++) {
            const segmentStart = this.routePath[i];
            const segmentEnd = this.routePath[i + 1];
            distance = this.distanceToSegment(position, segmentStart, segmentEnd);
            if (distance < deviationThreshold) {
                onRoute = true;
                break;
            }
        }
    }
    
    if (onRoute) {
      
      // Set the red color for the previous marker to be a standard marker
      new google.maps.Marker({
        position: this.marker.getPosition(),
        map: this.map,
        title: 'Previous Position',
        icon: {
          path: locationIconPath, // Use a location shape
          fillColor: 'green', // Change fillColor to green for previous positions
          fillOpacity: 1,
          scale: 2,
          strokeColor: 'white',
          strokeWeight: 2,
        },
      });
      

      // Marker is on the route, set to green
      this.marker.setIcon({
        path: this.marker.getIcon().path,
        fillColor: 'green',
        fillOpacity: 1,
        scale: 10,
        strokeColor: 'white',
        strokeWeight: 2,
      });
    } 
    else {

      // Set the red color for the previous marker to be a standard marker
      new google.maps.Marker({
        position: this.marker.getPosition(),
        map: this.map,
        title: 'Previous Position',
        icon: {
          path: locationIconPath, // Use a location shape
          fillColor: 'red', // Change fillColor to red for previous positions
          fillOpacity: 1,
          scale: 2,
          strokeColor: 'white',
          strokeWeight: 2,
        },
      });

      // Marker is off the route, set to red
      this.marker.setIcon({
        path: this.marker.getIcon().path,
        fillColor: 'red',
        fillOpacity: 1,
        scale: 10,
        strokeColor: 'white',
        strokeWeight: 2,
      });

      if(reachedOrigin){
        alert('Marker is out of the deviation!');
      }
    }
    
    setTimeout(() => {}, 1000);
  }

  updatePolylineColor(position: any) {
    console.log("reached on the route");
    const traveledPath = [];
    const untraveledPath = [];

    let hasReached = false;
    for (const point of this.routePath) {
      if (!hasReached && this.haversineDistance(position, point) < 1) {
        hasReached = true;
      }

      if (hasReached) {
        untraveledPath.push(point);
      } else {
        traveledPath.push(point);
      }
    }

    const traveledPolyline = new google.maps.Polyline({
      path: traveledPath,
      strokeColor: '#00FF00',
      strokeOpacity: 1.0,
      strokeWeight: 3,
    });

    const untraveledPolyline = new google.maps.Polyline({
      path: untraveledPath,
      strokeColor: '#FF0000',
      strokeOpacity: 1.0,
      strokeWeight: 3,
    });

    traveledPolyline.setMap(this.map);
    untraveledPolyline.setMap(this.map);
  }

  distanceToSegment(point: any, segmentStart: any, segmentEnd: any) {
    const toRadians = (degrees: number) => degrees * Math.PI / 180;
    const lat1 = toRadians(segmentStart.lat);
    const lng1 = toRadians(segmentStart.lng);
    const lat2 = toRadians(segmentEnd.lat);
    const lng2 = toRadians(segmentEnd.lng);
    const lat3 = toRadians(point.lat);
    const lng3 = toRadians(point.lng);

    const xDelta = lng2 - lng1;
    const yDelta = lat2 - lat1;

    const u = ((lng3 - lng1) * xDelta + (lat3 - lat1) * yDelta) / (xDelta * xDelta + yDelta * yDelta);

    const closestPoint = (() => {
      if (u < 0) return segmentStart;
      if (u > 1) return segmentEnd;
      return {
        lat: segmentStart.lat + u * (segmentEnd.lat - segmentStart.lat),
        lng: segmentStart.lng + u * (segmentEnd.lng - segmentStart.lng),
      };
    })();

    const haversineDistance = (pointA: any, pointB: any) => {
      const dLat = toRadians(pointB.lat - pointA.lat);
      const dLng = toRadians(pointB.lng - pointA.lng);
      const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
                Math.cos(toRadians(pointA.lat)) * Math.cos(toRadians(pointB.lat)) *
                Math.sin(dLng / 2) * Math.sin(dLng / 2);
      const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
      return 6371e3 * c;
    };

    return haversineDistance(point, closestPoint);
  }

  haversineDistance(coord1: any, coord2: any) {
    const toRadians = (degrees: number) => degrees * Math.PI / 180;
    const R = 6371e3; // Earth's radius in meters
    const lat1 = toRadians(coord1.lat);
    const lat2 = toRadians(coord2.lat);
    const deltaLat = toRadians(coord2.lat - coord1.lat);
    const deltaLng = toRadians(coord2.lng - coord1.lng);

    const a = Math.sin(deltaLat / 2) * Math.sin(deltaLat / 2) +
              Math.cos(lat1) * Math.cos(lat2) *
              Math.sin(deltaLng / 2) * Math.sin(deltaLng / 2);
    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));

    return R * c;
  }

  takePicture = async (card: number) => {
    const image = await Camera.getPhoto({
      quality: 90,
      allowEditing: false,
      resultType: CameraResultType.Uri,
      source: CameraSource.Camera,
      saveToGallery: true,
    });

    this.imageSource[card] = this.domSanitizer.bypassSecurityTrustUrl(
      image.webPath ? image.webPath : ''
    );

    // Convert the captured image to a Blob
    const response = await fetch(image.webPath!);
    const blob = await response.blob();
    this.file = new File([blob], `image-${card}.jpg`, { type: 'image/jpeg' });
  };
