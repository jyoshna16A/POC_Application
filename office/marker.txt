  initMap() {
    // Define the origin and destination points
    const origin = new google.maps.LatLng(this.pickuplng, this.pickuplt);
    const destination = new google.maps.LatLng(this.droplng, this.droplt);

    // Create a new map instance
    this.map = new google.maps.Map(document.getElementById('map'), {
        center: origin,
        zoom: 8,
        mapTypeId: google.maps.MapTypeId.ROADMAP,
    });

    // Create a DirectionsService object to handle directions requests
    const directionsService = new google.maps.DirectionsService();

    // Create a DirectionsRenderer object to display the directions
    const directionsRenderer = new google.maps.DirectionsRenderer({
        map: this.map,
    });

    // Request and display directions from the origin to the destination
    directionsService.route({
        origin: origin,
        destination: destination,
        travelMode: google.maps.TravelMode.DRIVING,
    }, (response : any, status :  any) => {
        if (status === google.maps.DirectionsStatus.OK) {
            directionsRenderer.setDirections(response);
          } else {
            console.error('Directions request failed due to ' + status);
        }

            // Get the polyline representing the route
            const route = response.routes[0].overview_path;

            // Create a polyline from the route
            const routePolyline = new google.maps.Polyline({
                path: route,
                geodesic: true,
                strokeColor: '#FF0000',
                strokeOpacity: 1.0,
                strokeWeight: 2,
            });

            routePolyline.setMap(this.map);

            // Create a 5 km rectangular buffer around the origin and destination
            const bufferDistance = 5000; // 5 km
            let minLat = Math.min(origin.lat(), destination.lat());
            let maxLat = Math.max(origin.lat(), destination.lat());
            let minLng = Math.min(origin.lng(), destination.lng());
            let maxLng = Math.max(origin.lng(), destination.lng());

            // Extend the bounding box by the buffer distance
            const bufferLat = bufferDistance / 111320; // Convert meters to degrees latitude (approximately)
            const bufferLng = bufferDistance / (111320 * Math.cos(minLat * (Math.PI / 180))); // Convert meters to degrees longitude (approximately)

            const bounds = new google.maps.LatLngBounds(
                new google.maps.LatLng(minLat - bufferLat, minLng - bufferLng),
                new google.maps.LatLng(maxLat + bufferLat, maxLng + bufferLng)
            );

            // Display the buffer as a rectangle
            this.bufferRectangle = new google.maps.Rectangle({
                bounds: bounds,
                strokeColor: 'red',
                strokeOpacity: 0.2,
                strokeWeight: 1,
                fillColor: 'red',
                fillOpacity: 0.1,
            });

            this.bufferRectangle.setMap(this.map);

            // Initialize the marker at the pickup location with bouncing animation
            this.marker = new google.maps.Marker({
                position: { lat: this.pickuplt, lng: this.pickuplng },
                map: this.map,
                title: 'Current Position',
                animation: google.maps.Animation.BOUNCE,
                icon: {
                    path: this.locationIconPath,
                    fillColor: this.bufferRectangle.getBounds().contains(new google.maps.LatLng(this.pickuplt, this.pickuplng)) ? 'green' : 'red',
                    fillOpacity: 1,
                    scale: 2,
                    strokeColor: 'white',
                    strokeWeight: 2,
                },
            });
          

            // Center the map to the initial marker position
            this.map.setCenter(new google.maps.LatLng(this.pickuplt, this.pickuplng));
          });
            // Place additional markers from gpsCordsList
            this.gpsCordsList.forEach(cord => {
              console.log("gpsCordsList")
              new google.maps.Marker({
                position: { lat: cord.latitude, lng: cord.longitude },
                map: this.map,
                title: 'Updated Position',
                icon:{
                  path : this.locationIconPath,
                  fillColor : 'green',
                  fillOpacity : 1,
                  scale : 2,
                  strokeColor : 'white',
                  strokeWeight : 2,
                }
              });
            
    });

    this.locationIconPath = 'M12 2C8.13 2 5 5.13 5 9c0 5.25 7 13 7 13s7-7.75 7-13c0-3.87-3.13-7-7-7zm0 9.5c-1.38 0-2.5-1.12-2.5-2.5S10.62 6.5 12 6.5 14.5 7.62 14.5 9 13.38 11.5 12 11.5z M10 7h4v5h-4z M11 14h2v1h-2z'; // Simplified and adjusted train icon path
}

updateMarkerPosition(newLat :  any, newLng :  any) {
     // Update the current marker position with bouncing animation
     const newPosition = new google.maps.LatLng(newLat, newLng);
     this.marker.setPosition(newPosition);
     this.marker.setAnimation(google.maps.Animation.BOUNCE);
 
     // Center the map to the new marker position
     this.map.setCenter(newPosition);
    // Check if the new position is within the buffer rectangle
    if (this.bufferRectangle.getBounds().contains(newPosition)) {
        this.marker.setIcon({
            path: this.locationIconPath,
            fillColor: 'green',
            fillOpacity: 1,
            scale: 2,
            strokeColor: 'white',
            strokeWeight: 2,
        });
    } else {
        this.marker.setIcon({
            path: this.locationIconPath,
            fillColor: 'red',
            fillOpacity: 1,
            scale: 2,
            strokeColor: 'white',
            strokeWeight: 2,
        });
    }
    // this.getAllGPSCords1();
   
}

  startGeolocationUpdates() {
    if (this.trackStatus === 'Pick Up Station') {
      this.locationInterval = setInterval(() => {
        this.geolocation
          .getCurrentPosition()
          .then((resp) => {
            this.Latitude = resp.coords.latitude;
            this.Longitude = resp.coords.longitude;

            console.log('lat long', this.Latitude, this.Longitude);

            this.trackService
              .updatecords(this.id, this.Latitude, this.Longitude)
              .subscribe((data) => {
                console.log('data response', data);
              });
            this.trackService
              .updatecords1(this.id, this.Latitude, this.Longitude)
              .subscribe((data) => {
                console.log('data response1', data);
              });

            this.trackService
              .getLatestCoordinates(this.id)
              .subscribe((data: any) => {
                console.log('data response3', data);
                this.updateMarkerPosition(data.latitude, data.longitude);
              });
          })
          .catch((error) => {
            console.log('Error getting location', error);
          });
      }, 3000);
    } else {
      console.log('Closed');
    }
  }

  stopGeolocationUpdates() {
    clearInterval(this.locationInterval);

  updateStatus(status: any) {
    if (status == 'pickup') {
      this.trackStatus = 'Pick Up Station';
      this.startGeolocationUpdates();
      this.trackService
        .updatestatus(this.id, this.trackStatus)
        .subscribe((data) => {
          console.log('data response', data);
        });
    } else if (status == 'transit') {
      console.log('Transit');
      this.trackStatus = 'On the way';
      this.trackService
        .updatestatus(this.id, this.trackStatus)
        .subscribe((data) => {
          console.log('data response', data);
        });
    } else if (status == 'delivered') {
      console.log('Delivered');
      // this.stopGeolocationUpdates();
      this.trackStatus = 'Delivery Station';
      this.trackService
        .updatestatus(this.id, this.trackStatus)
        .subscribe((data) => {
          console.log('data response', data);
        });
    } else {
      console.log('Ready for pickup');
      this.trackStatus = 'Ready for Pickup';
      this.trackService
        .updatestatus(this.id, this.trackStatus)
        .subscribe((data) => {
          console.log('data response', data);
        });
      this.trackService
        .updateTrainCompletedStatus(this.id)
        .subscribe((data) => {
          console.log(data);
        });
    }
  }

  takePicture = async (card: number) => {
    const image = await Camera.getPhoto({
      quality: 90,
      allowEditing: false,
      resultType: CameraResultType.Uri,
      source: CameraSource.Camera,
      saveToGallery: true,
    });

    this.imageSource[card] = this.domSanitizer.bypassSecurityTrustUrl(
      image.webPath ? image.webPath : ''
    );

    // Convert the captured image to a Blob
    const response = await fetch(image.webPath!);
    const blob = await response.blob();
    this.file = new File([blob], `image-${card}.jpg`, { type: 'image/jpeg' });
  };

  getPhoto(cardIndex: number) {
    return this.selectedCard === cardIndex
      ? this.imageSource[cardIndex]
      : undefined;
  }

  toggleImage(card: number) {
    if (this.selectedCard === card) {
      this.selectedCard = null;
    } else {
      this.selectedCard = card;
    }
  }

    const toast = await this.toastController.create({
      message: message,

      color: color,
      position: 'top',

  }

  onSubmit() {
    if (this.file) {
      const formData: FormData = new FormData();
      formData.append('id', this.id.toString());
      formData.append('image', this.file, this.file.name);
      console.log(this.id, 'thisid');
      formData.forEach((value, key) => {
        console.log(`${key}: ${value}`);
      });
      this.trackService.addMedia(formData).subscribe(
        (response) => {
          console.log('Image uploaded successfully', response);
          this.presentToast('Image uploaded successfully', 'success');
        },
        (error) => {
          console.error('Error uploading image:', error);
          this.presentToast('Error uploading image', 'danger');
        }
      );
    } else {
      console.error('No files selected');
      this.presentToast('No files selected', 'danger');
    }
  }
